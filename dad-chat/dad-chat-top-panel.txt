// ── Dynamic theme reader for plugin styling ──
// Reads computed CSS vars from :root (dark) or body[data-theme="light"] (light)
// and returns style strings that comments-plugin & the public-gallery embed can consume.
getThemeCssVar(name) =>
  return getComputedStyle(document.body).getPropertyValue(name).trim();

themeStyle(kind) =>
  let bgMain = getThemeCssVar('--bg-main');
  let bgSec = getThemeCssVar('--bg-sec');
  let bgTer = getThemeCssVar('--bg-ter');
  let textMain = getThemeCssVar('--text-main');
  let textMuted = getThemeCssVar('--text-muted');
  let accent = getThemeCssVar('--accent');
  let danger = getThemeCssVar('--danger');
  let success = getThemeCssVar('--success');
  let isLight = document.body.getAttribute('data-theme') === 'light';
  let bgInput = isLight ? '#ffffff' : bgSec;
  let border = isLight ? bgTer : bgTer;
  let shadow = isLight ? '0 4px 20px rgba(0,0,0,0.06)' : '0 6px 24px rgba(0,0,0,0.3)';
  switch(kind) {
    case "container":
      return `background:${bgSec}; border:1px solid ${border}; box-shadow:${shadow}; width:100%; max-width:1024px; height:min(600px, 70vh); max-height:80vh; padding:16px; margin:auto; border-radius:10px;`;
    case "bubble":
      return `background:${bgMain}; color:${textMain}; border:1px solid ${border}; border-radius:10px; padding:10px 14px; margin-bottom:9px; font-family:'Inter',sans-serif; font-size:0.8rem; word-break:break-word; overflow-wrap:anywhere;`;
    case "feed":
      return `background:${bgMain}; padding:10px; border-radius:6px;`;
    case "input":
      return `background:${bgInput}; border:1px solid ${border}; border-radius:8px; color:${textMain}; font-size:15px; padding:8px 12px;`;
    case "submit":
      return `background:${accent}; color:${isLight ? '#ffffff' : bgMain}; border:none; border-radius:4px; font-family:'Inter',sans-serif; font-weight:700; box-shadow:0 2px 8px rgba(0,0,0,0.2); transition:filter .2s; cursor:pointer; padding:6px 14px; margin:0.2rem;`;
    case "gallery":
      return `background:${bgSec}; border:1px solid ${border}; border-radius:10px; padding:12px; max-width:1024px; margin:0 auto; box-shadow:${shadow};`;
  }
  return "";

// Submit button style for the forum embed: same as themeStyle("submit") but with a

// ── Public gallery embed (replaces the official plugin's image({gallery:true})) ──
// the image-plugin only does generation, so the public-gallery iframe is built here
// with the exact same protocol as the official text-to-image-plugin: same URL, same
// #data hash carrying the moderation lists (bannedUsers/bannedPromptPhrases) and the
// themed background, same .text-to-image-plugin-gallery class so the app's existing
// render/cleanup hooks and the adaptive-height bridge keep working.
galleryEmbedHtml() =>
  let opts = galleryOptions;
  if(!opts) return "";
  let sortVal = "recent";
  let timeRangeVal = "1-week";
  let contentFilterVal = "pg13";
  let hideScore = -2;
  let styleStr = "";
  try {
    sortVal = opts.sort ? ("" + opts.sort.evaluateItem) : "recent";
    if(sortVal === "best") sortVal = "top";
    timeRangeVal = opts.timeRange ? ("" + opts.timeRange.evaluateItem) : "1-week";
    contentFilterVal = opts.contentFilter ? ("" + opts.contentFilter.evaluateItem).toLowerCase() : "pg13";
    if(contentFilterVal === "pg-13") contentFilterVal = "pg13";
    if(contentFilterVal !== "pg13" && contentFilterVal !== "g") contentFilterVal = "g";
    hideScore = Number(opts.hideIfScoreIsBelow && opts.hideIfScoreIsBelow.evaluateItem);
    if(isNaN(hideScore)) hideScore = -2;
    styleStr = opts.style ? ("" + opts.style.evaluateItem) : "";
  } catch(e) {}
  let bannedUsers = [];
  try { if(opts.bannedUsers && opts.bannedUsers.selectAll) bannedUsers = opts.bannedUsers.selectAll.map(i => "" + i.evaluateItem); } catch(e) {}
  let bannedPromptPhrases = [];
  try {
    if(opts.bannedPromptPhrases && opts.bannedPromptPhrases.selectAll) {
      bannedPromptPhrases = opts.bannedPromptPhrases.selectAll.map(i =>
        (i.getRawListText !== undefined ? i.getRawListText : ("" + i.evaluateItem)).replace("/\\^", "/^")
      );
    }
  } catch(e) {}
  let injectedStyles = {};
  let bgMatch = styleStr.match(/(?:^|;) *background(?:-color)?:(.+?)(?:;|$)/);
  if(bgMatch) injectedStyles.background = bgMatch[1].trim();
  let hashData = {
    galleryInstanceId: "dadGallery" + Math.random().toString(36).slice(2),
    shouldShowCustomButton: false,
    customButtonEmoji: null,
    shouldShowCustomButton2: false,
    customButton2Emoji: null,
    bannedUsers: bannedUsers,
    bannedPromptPhrases: bannedPromptPhrases,
    bannedNegativePromptPhrases: [],
    injectedStyles: injectedStyles,
    defaultSubChannelNames: []
  };
  let galleryOptionsData = { sort: sortVal, timeRange: timeRangeVal, hideIfScoreIsBelow: hideScore, contentFilter: contentFilterVal, subChannel: "public" };
  let url = "https://image-generation.perchance.org/gallery"
    + "?channel=" + encodeURIComponent(window.generatorName)
    + "&sort=" + encodeURIComponent(sortVal)
    + "&timeRange=" + encodeURIComponent(timeRangeVal)
    + "&hideIfScoreIsBelow=" + hideScore
    + "&contentFilter=" + encodeURIComponent(contentFilterVal)
    + "&subChannel=public"
    + "#data=" + encodeURIComponent(JSON.stringify(hashData));
  return `<iframe class="text-to-image-plugin-gallery" data-gallery-options="${encodeURIComponent(JSON.stringify(galleryOptionsData))}" data-adaptive-height="yes" style="width:100%; height:70vh; border:none; ${styleStr}" src="${url}" allow="clipboard-write"></iframe>`;
